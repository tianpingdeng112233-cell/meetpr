import AVFoundation
import Foundation
import Security

let probeID = "meetpr.spec110.diagnostic.\(UUID().uuidString)"
let query: [String: Any] = [
  kSecClass as String: kSecClassGenericPassword,
  kSecAttrService as String: probeID,
  kSecAttrAccount as String: "nonexistent-diagnostic-item",
  kSecMatchLimit as String: kSecMatchLimitOne,
  kSecReturnData as String: true
]
var item: CFTypeRef?
let status = SecItemCopyMatching(query as CFDictionary, &item)
print("Keychain unique nonexistent-item query status:", status)
print("Keychain status description:", SecCopyErrorMessageString(status, nil) as String? ?? "unknown")
let output = URL(fileURLWithPath: CommandLine.arguments[1])
let writer = try AVAssetWriter(outputURL: output, fileType: .mov)
let input = AVAssetWriterInput(mediaType: .video, outputSettings: [
  AVVideoCodecKey: AVVideoCodecType.h264,
  AVVideoWidthKey: 64,
  AVVideoHeightKey: 64
])
writer.add(input)
print("Standalone H.264 writer.startWriting:", writer.startWriting())
if let error = writer.error as NSError? {
  print("Writer error domain/code:", error.domain, error.code)
  print("Writer error details:", error.userInfo)
}
writer.cancelWriting()
try? FileManager.default.removeItem(at: output)
